# `toolchain`, `zlib`, `expat` and `libdrm` come from the constants file
# build.sh generates and passes as the FIRST --cross-file: machine-
# specific paths ($NEOOS_TOOLCHAIN, $ZLIB_DIR, ...) never live here.
[binaries]
c = toolchain / 'bin/x86_64-neoos-linux-musl-gcc'
cpp = toolchain / 'bin/x86_64-neoos-linux-musl-g++'
ar = toolchain / 'bin/x86_64-neoos-linux-musl-gcc-ar'
strip = toolchain / 'bin/x86_64-neoos-linux-musl-strip'
pkg-config = '/usr/bin/pkg-config'

[host_machine]
system = 'linux'
cpu_family = 'x86_64'
cpu = 'x86_64'
endian = 'little'

[properties]
# Without this, Meson's pkg-config lookups for a cross build still see
# the HOST's default pkg-config search paths, silently resolving
# zlib/expat (etc.) against glibc-ABI host packages instead of failing
# or finding the NeoOS-cross ones -- found the hard way during this
# port's own bring-up (Meson reported "Run-time dependency zlib found:
# YES 1.3.1" using the host's system zlib before this was added).
# Setting pkg_config_libdir explicitly is Meson's documented fix: it
# replaces the default search dirs entirely for this cross build, so
# only packages actually built for x86_64-neoos-linux-musl are ever
# found. Deliberately NOT setting sys_root: that property makes Meson
# prepend it to every path pkg-config reports (via
# PKG_CONFIG_SYSROOT_DIR), which is only correct for .pc files staged
# under a real sysroot tree -- ours report their own absolute
# build-output paths directly, and sys_root corrupted them into
# nonexistent concatenated paths (found the hard way: "zlib.h: No such
# file or directory" with an include path of
# ".../x86_64-neoos-linux-musl/home/neo/projects/personal/neoos-zlib/...").
pkg_config_libdir = [zlib / 'lib/pkgconfig', expat / 'lib/pkgconfig', libdrm / 'lib/pkgconfig']
